% Wall time vs. CPU time. Wall time is the span from the first rank starting to
% the last one finishing; CPU time is the sum of the individual bars, which is
% why it grows with the degree of parallelism even when wall time shrinks.
\documentclass[dvisvgm,border=3pt]{standalone}
\input{preamble.tex}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  \def\barh{0.52}

  % \rankbar{row}{start}{end}{label}
  \newcommand{\rankbar}[4]{%
    \fill[uibkorange,draw=uibkorange!70!black,line width=0.5pt]
      (#2,{#1*0.80+0.20}) rectangle (#3,{#1*0.80+0.20+\barh});
    \node[text=inkdark,font=\sffamily\small]
      at ({(#2+#3)/2},{#1*0.80+0.20+\barh/2}) {#4};
    % guides marking when this rank starts and stops
    \draw[draw=uibknavy!40!blu,line width=0.9pt,dash pattern=on 4pt off 3pt]
      (#2,0) -- (#2,2.75);
    \draw[draw=uibknavy!40!blu,line width=0.9pt,dash pattern=on 4pt off 3pt]
      (#3,0) -- (#3,2.75);}

  \rankbar{0}{0.00}{4.30}{R0}
  \rankbar{1}{0.95}{5.55}{R1}
  \rankbar{2}{0.45}{4.95}{R2}

  % axes
  \draw[->,draw=blu,line width=1.2pt] (0,0) -- (0,3.15);
  \draw[->,draw=blu,line width=1.2pt] (0,0) -- (6.10,0);
  \node[text=fg,font=\sffamily\small,anchor=south east] at (0.25,3.20) {ranks};
  \node[text=fg,font=\sffamily\small,anchor=west] at (6.15,0) {time};

\end{tikzpicture}
\end{document}
