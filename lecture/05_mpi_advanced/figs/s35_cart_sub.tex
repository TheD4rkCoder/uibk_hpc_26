% MPI_Cart_sub cuts a Cartesian communicator into slices; each slice becomes a
% communicator of its own, so a collective on A leaves B, C and D untouched.
\documentclass[dvisvgm,border=3pt]{standalone}
\input{preamble.tex}
\input{_cart.tex}
\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

  \carttopo{0/0/h, 1/0/h, 2/0/h, 3/0/h, 0/1/h, 1/1/h, 2/1/h, 3/1/h,
            0/2/h, 1/2/h, 2/2/h, 3/2/h, 0/3/h, 1/3/h, 2/3/h, 3/3/h}

  % every rank is labelled with the slice it now belongs to
  \foreach \r/\l in {0/A, 1/B, 2/C, 3/D}{
    \foreach \c in {0,...,3}{
      \node[text=inkdark,font=\sffamily\scriptsize] at ({\c*\cartd},{-\r*\cartd}) {\l};}}

  % the cuts between the slices
  \foreach \r in {0,1,2}{
    \draw[draw=blu,line width=1.4pt,dash pattern=on 6pt off 4pt]
      (-0.62,{-\r*\cartd-0.5*\cartd}) -- ({3*\cartd+0.62},{-\r*\cartd-0.5*\cartd});}

\end{tikzpicture}
\end{document}
