% The three steps, as pictures: the problem domain, the same domain cut into
% pieces (not necessarily regular ones), and the communication and
% synchronisation that the cut then forces upon you.
\documentclass[dvisvgm,border=3pt]{standalone}
\input{preamble.tex}
\begin{document}
\begin{tikzpicture}[
    x=1cm,y=1cm,
    cell/.style ={draw=blu,line width=0.5pt},
    part/.style ={draw=uibkorange,line width=1.4pt},
    dot/.style  ={circle,fill=blu,inner sep=0pt,minimum size=2.4mm},
    thin arr/.style={->,draw=blu,line width=0.4pt,
                     >={Stealth[round,length=2.4pt,width=2.2pt]}},
    pan/.style  ={text=muted,font=\sffamily\scriptsize,align=center}]

  \def\g{0.30}   % grid cell size

  % =============== 1. the data ==========================================
  \begin{scope}[xshift=0cm]
    \foreach \c in {0,...,5}{\foreach \r in {0,...,5}{
      \draw[cell] ({\c*\g},{\r*\g}) rectangle ++(\g,\g);}}
    \node[pan] at (0.90,-0.45) {identify};
  \end{scope}

  % =============== 2. the partitioning ==================================
  \begin{scope}[xshift=2.6cm]
    \foreach \c in {0,...,5}{\foreach \r in {0,...,5}{
      \draw[cell] ({\c*\g},{\r*\g}) rectangle ++(\g,\g);}}
    % irregular blocks: three strips, an L and a diamond
    \foreach \x in {0.30,0.60}{\draw[part] (\x,0) -- (\x,0.90);}
    \draw[part] (0,0.90) -- (0.90,0.90);
    \draw[part] (1.20,1.80) -- (1.80,1.80) -- (1.80,0.90)
             -- (1.20,0.90) -- cycle;
    \draw[part] (1.35,0.84) -- (1.74,0.45) -- (1.35,0.06) -- (0.96,0.45) -- cycle;
    \node[pan] at (0.90,-0.45) {partition};
  \end{scope}

  % =============== 3. communication + synchronisation ===================
  \begin{scope}[xshift=5.4cm,yshift=0.9cm]
    \foreach \i in {0,...,7}{
      \node[dot] (p\i) at ({45*\i}:0.78) {};}
    \foreach \i in {0,...,7}{\foreach \j in {0,...,7}{
      \ifnum\i<\j \draw[thin arr] (p\i) -- (p\j);\fi}}
    \node[pan] at (0,-1.35) {communicate};
  \end{scope}

  \begin{scope}[xshift=7.9cm]
    \draw[tline,line width=0.9pt] (0,1.80) -- (0,0);
    \draw[tline,line width=0.9pt] (0.85,1.80) -- (0.85,0);
    \draw[->,draw=uibkorange,line width=0.9pt] (0,1.45) -- (0.85,1.15);
    \draw[->,draw=uibkorange,line width=0.9pt] (0.85,0.95) -- (0,0.65);
    \node[pan] at (0.42,-0.45) {synchronise};
  \end{scope}

\end{tikzpicture}
\end{document}
